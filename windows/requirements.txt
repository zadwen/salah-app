# Windows build dependencies.
# tkinter ships with the standard python.org Windows installer already
# (no pip package needed) -- if "import tkinter" fails, reinstall Python
# and make sure "tcl/tk and IDLE" is checked in the installer.

Pillow>=10.0          # tray/app icon generation
pystray>=0.19          # system tray icon + menu
pygame>=2.5.0          # notification sound playback (wav/mp3/ogg) with a real stop() --
                       # needed so Mute/closing a notification can actually cut off sound
                       # already in progress, not just block future sounds.

# Optional, best-effort only -- the app works fine without it:
plyer>=2.1              # bonus native Windows toast (falls back silently if missing)

pyinstaller>=6.0        # only needed to build the standalone .exe yourself
